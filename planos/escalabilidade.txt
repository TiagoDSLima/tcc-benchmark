# Escalabilidade, sempre com o teste de estresse do cenário misto.
# Ajuste TAXA depois da calibração para que o último degrau passe do ponto de saturação.
# Rodar: bash scripts/lote.sh planos/escalabilidade.txt
# Duração aproximada com 5 repetições: 8 h.

# Vertical: mesmo container(s), com 2, 4 e 8 CPUs na camada de aplicação (memória fixa)
misto estresse TAXA=400 VUS_MAX=1000 DURACAO_PASSO=30s RECURSOS=2cpu
misto estresse TAXA=400 VUS_MAX=1000 DURACAO_PASSO=30s RECURSOS=4cpu
misto estresse TAXA=600 VUS_MAX=1000 DURACAO_PASSO=30s RECURSOS=8cpu

# Horizontal: balanceador nginx na frente das réplicas. A linha com 1 réplica é a referência,
# porque também passa pelo balanceador.
misto estresse TAXA=400 VUS_MAX=1000 DURACAO_PASSO=30s ESCALA=1
misto estresse TAXA=600 VUS_MAX=1000 DURACAO_PASSO=30s ESCALA=1 ARQ=monolito REPLICAS_MONOLITO=2
# Nos microsserviços, replique os serviços que mais consumiram CPU na calibração (exemplo abaixo)
misto estresse TAXA=600 VUS_MAX=1000 DURACAO_PASSO=30s ESCALA=1 ARQ=microsservicos REPLICAS_GATEWAY=2 REPLICAS_PRODUTO=2 REPLICAS_PEDIDO=2
